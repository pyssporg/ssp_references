# Minimal environment for examples/simple_chain.
#
# Run from the REPOSITORY ROOT so the relative paths below resolve:
#
#   python3 -m venv examples/simple_chain/build/venv
#   examples/simple_chain/build/venv/bin/pip install -r examples/simple_chain/requirements.txt
#   examples/simple_chain/build/venv/bin/python examples/simple_chain/run.py
#
# The packages the example imports come from their GitHub release assets, so the
# demo is reproducible and does not depend on a local checkout. Each wheel names
# its sibling dependencies the same way, so the tags must be released.

# The experimental pyssp_standard fork. Its GitHub repository is named
# `experimental_pyssp_standard`, the distribution is `experimental-pyssp-standard`,
# and the import package stays `pyssp_standard`. pyssp_sysml2 needs its refactored API,
# and the PyPI release (0.9.1) predates it.
experimental-pyssp-standard @ https://github.com/pyssporg/experimental_pyssp_standard/releases/download/v0.9.2.dev0/experimental_pyssp_standard-0.9.2.dev0-py3-none-any.whl
pyssp-sysml2 @ https://github.com/pyssporg/pyssp_sysml2/releases/download/v0.3.0/pyssp_sysml2-0.3.0-py3-none-any.whl
pyssp-simulate @ https://github.com/pyssporg/pyssp_simulate/releases/download/v0.2.0/pyssp_simulate-0.2.0-py3-none-any.whl

# ssp4sim backend. Not on PyPI; the wheel is Linux x86_64 only, so it is
# pinned here exactly as the repository-level requirements.txt pins it.
https://github.com/jkCXf9X4/ssp4sim/releases/download/v0.3.2/pyssp4sim-0.3.2-cp39-abi3-linux_x86_64.whl
