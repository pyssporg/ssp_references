# Minimal environment for examples/simple_chain.
#
# Run from the REPOSITORY ROOT so the relative paths below resolve:
#
#   python3 -m venv examples/simple_chain/build/venv
#   examples/simple_chain/build/venv/bin/pip install -r examples/simple_chain/requirements.txt
#   examples/simple_chain/build/venv/bin/python examples/simple_chain/run.py
#
# The packages the example imports are installed from their GitHub repositories
# at the commits the repository's vendored checkouts currently use, so the demo
# is reproducible and does not depend on a local checkout.

# The experimental pyssp_standard checkout. Its GitHub repository is named
# `experimental_pyssp_standard`; the distribution and import name remain
# `pyssp-standard` / `pyssp_standard`. pyssp_sysml2 needs its refactored API,
# and the PyPI release (0.9.1) predates it, so install this first.
pyssp-standard @ git+https://github.com/pyssporg/experimental_pyssp_standard.git@676673b81fb1d5802d4728392c229ef412b01835
pyssp_sysml2 @ git+https://github.com/pyssporg/pyssp_sysml2.git@4ebc925437b0bd7fa03f187aef2eb116e29c232b
pyssp_simulate @ git+https://github.com/pyssporg/pyssp_simulate.git@b7b0a61cd6a02672a600422a8f48ec39fcbea12b

# ssp4sim backend. Not on PyPI; the wheel is Linux x86_64 only, so it is
# pinned here exactly as the repository-level requirements.txt pins it.
